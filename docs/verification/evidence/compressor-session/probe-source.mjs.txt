const errorText = result => (result?.model?.[0]?.content ?? []).filter(x => x.type === 'text').map(x => x.text).join('\n');

async function fixture(name, reply, { policy = false, onYield, afterResponse, rawGuard = false, repairErrors = false } = {}) {
 const directory = join(out, name); await mkdir(directory, { recursive: true });
 const requests = [], events = [], reports = [], executed = []; const rawArguments = new Map();
 const completeRewrite = 'REWRITE ENTIRE submit_compaction: summary, graphs including annotations, clues and unresolved. Do not submit a partial patch.';
 const apiImplementation = rawGuard ? { ...oac, streamSimple: (model, transcript, options) => oac.streamSimple(model, transcript, { ...options, onProviderStreamEvent: async (data, currentModel) => { await options?.onProviderStreamEvent?.(data, currentModel); for (const choice of data.choices ?? []) for (const call of choice.delta?.tool_calls ?? []) { const key = call.id ?? 'index-' + call.index; const state = rawArguments.get(key) ?? { text: '', id: call.id }; state.text += call.function?.arguments ?? ''; rawArguments.set(key, state); } } }) } : oac;

 const server = createServer(async (req, res) => {
  try {

   const n = requests.length; const answer = await reply(n, payload);
   if (answer.http) { res.writeHead(answer.http, { 'content-type': 'application/json' }); res.end(JSON.stringify({ error: { message: 'provider-failure-' + n } })); return; }
   const base = { id: 'probe-' + n, object: 'chat.completion.chunk', created: 1, model: 'local-model' };
   const delta = answer.tool ? { tool_calls: [{ index: 0, id: 'call-' + n, type: 'function', function: { name: answer.tool, arguments: answer.argsJson ?? JSON.stringify(answer.args ?? {}) } }] } : { content: answer.text ?? 'ordinary-stop-' + n };
   const chunks = [ { ...base, choices: [{ index: 0, delta: { role: 'assistant' }, finish_reason: null }] }, { ...base, choices: [{ index: 0, delta, finish_reason: null }] }, { ...base, choices: [{ index: 0, delta: {}, finish_reason: answer.finishReason ?? (answer.tool ? 'tool_calls' : 'stop') }], usage: { prompt_tokens: 10, completion_tokens: 2, total_tokens: 12 } } ];
   res.writeHead(200, { 'content-type': 'text/event-stream' }); for (const chunk of chunks) res.write('data: ' + JSON.stringify(chunk) + '\n\n'); res.end('data: [DONE]\n\n');
  } catch (error) { reports.push('server:' + String(error)); res.destroy(error); }
 });

 }

   onYield: async (message, api, ctx) => { events.push({ event: 'onYield', stopReason: message.stopReason }); if (policy) { const doc = await bump(api.conversationId, 'unresolvedErrors', 'no-complete-submission:' + message.stopReason + ':' + requests.length, 'yield:' + api.taskId); if (doc.status === 'running') return { continue: 'No complete accepted submission.    afterTools: async (assistant, ids, api, ctx) => { const records = await harness.commit(tx => Promise.all(ids.map(id => tx.entry(ToolResultEntry, id))), ctx); const errors = records.filter(record => record?.model?.[0]?.isError); events.push({ event: 'afterTools', errors: errors.map(errorText), results: records.map(record => record?.model?.[0]) }); if (policy) for (const record of errors) await bump(api.conversationId, 'toolErrors', errorText(record), 'tool:' + record.id); if (repairErrors && errors.length) await harness.commit(async tx => { for (const record of errors) { const original = record.model[0]; await tx.appendEntry(api.conversationId, { kind: 'probe.full-rewrite-feedback', data: { target: record.id }, edits: [{ target: record.id, action: 'replace', messages: [{ ...original, content: [...original.content, { type: 'text', text: completeRewrite }] }] }] }); } }, ctx); }
  })
 ] }));
